  const clean=s=>String(s||'').replace(/\s+/g,' ').trim();
  const norm=s=>clean(s).replace(/[\s_-]+/g,' ').toUpperCase();
  function identify(rule){
    const owner=standardOwner(rule?.owner);
    if(owner!=='TMR'&&owner!=='QRL')return null;
    const raw=String(rule?.prop||'').split('|')[0].trim();
    const pos=raw.indexOf('.');
    if(pos<1)return null;
    const pset=clean(raw.slice(0,pos));
    // Owner Pset opt-outs can never suppress common DSCRLP project checks.
    if(!pset||pset==='*'||/^Attr$/i.test(pset)||/^DSCRLP[ _-]*PROJECT/i.test(pset))return null;
    return {owner,pset,key:owner+'::'+norm(pset)};
  }
  function stageItems(schema,gate){
    // Always list every TMR and QR Pset defined in the imported project LoIN,
    // including property sets not scheduled for the currently selected gate.
    // No fabricated Pset names or requirements.
    const all=new Map();
    for(const r of schema?.rules||[]){
      const v=identify(r);if(!v)continue;
      if(!all.has(v.key))all.set(v.key,{...v,required:0,placeholder:0});
      const mark=r.stages? r.stages[gate] : 'R';
      if(mark==='P')all.get(v.key).placeholder++;
      else if(mark==='R')all.get(v.key).required++;
    }
    const saved=schema?.settings?.[KEY]?.[gate]||{};
    return [...all.values()].map(x=>({
      ...x,applicable:(x.required+x.placeholder)>0,
      enabled:saved[x.key]!==false
    })).sort((a,b)=>a.owner.localeCompare(b.owner)||a.pset.localeCompare(b.pset));
  }
  function isExcluded(rule,gate,schema){
    const v=identify(rule);
    const mark=rule?.stages?rule.stages[gate]:'R';
    return !!(v&&(mark==='R'||mark==='P')&&schema?.settings?.[KEY]?.[gate]?.[v.key]===false);
  }
  // Used directly by the application validator. A rule's required stage and
  // source matrix markers are retained untouched for future submissions.
  window.slqGateExcluded=isExcluded;
  const htmlEscape=value=>String(value).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
  const currentGate=()=>document.getElementById('stageSel')?.value||state()?.schema?.settings?.stage||'';
  function render(){
    const root=document.getElementById('slqStageScope'),app=state();
    if(!root||!app)return;
    const schema=app.schema,gate=currentGate();
    if(!schema||!gate){root.hidden=true;return;}
    const items=stageItems(schema,gate);
    if(!items.length){root.hidden=true;root.innerHTML='';return;}
    root.hidden=false;
    const counts={TMR:items.filter(x=>x.owner==='TMR'),QRL:items.filter(x=>x.owner==='QRL')};
    const nDeferred=items.filter(x=>x.applicable&&!x.enabled).length;
    const nDue=items.filter(x=>x.applicable).length;
    const nNotDue=items.length-nDue;
    const reasons=schema.settings?.[REASONS]||{};
    root.innerHTML='<div class="slq-scope-head">TMR / QR property set delivery <span style="font-size:11px;opacity:.72">'+htmlEscape(gate)+'</span></div>'+
      '<div class="slq-scope-sub">All imported TMR and QR property sets are shown. Tick or untick those scheduled at the selected stage. Greyed-out sets are not due at this gate. Unchecked = deferred, not passed. Original LoIN unchanged.</div>'+
      ['TMR','QRL'].map(owner=>!counts[owner].length?'':'<div class="slq-scope-owner">'+(owner==='QRL'?'QR':owner)+'</div><div class="slq-scope-grid">'+counts[owner].map(item=>
        '<label class="slq-scope-option '+(!item.applicable?'notdue':item.enabled?'':'deferred')+'"><input type="checkbox" data-scope-key="'+encodeURIComponent(item.key)+'" '+(item.applicable&&item.enabled?'checked':'')+' '+(item.applicable?'':'disabled')+'><span><b>'+htmlEscape(item.pset)+'</b><small style="display:block;opacity:.72">'+(item.applicable?(item.required+' required · '+item.placeholder+' placeholder'):'Not scheduled at '+htmlEscape(gate))+'</small></span></label>').join('')+'</div>').join('')+
      '<div class="slq-scope-meta">'+items.length+' source Psets: '+(nDue-nDeferred)+' checked / '+nDeferred+' deferred / '+nNotDue+' not yet scheduled at '+htmlEscape(gate)+'. DSCRLP – PROJECT requirements remain compulsory.</div>'+
      (nDeferred?'<div class="slq-scope-note"><label for="slq-scope-reason">Reason for deferral (recorded in MVC)</label><textarea id="slq-scope-reason" placeholder="e.g. asset owner information not established for this stage">'+htmlEscape(reasons[gate]||'')+'</textarea></div>':'')+
      '<div class="slq-scope-actions"><span class="slq-scope-warn">A deferred property set is excluded from the compliance percentage; formal approval is separate.</span><button type="button" id="slq-scope-reset">Enable all</button></div>';
    root.querySelectorAll('[data-scope-key]').forEach(box=>{
      box.onchange=()=>{
        const key=decodeURIComponent(box.dataset.scopeKey);
        const current=schema.settings?.[KEY]||{};
        const forGate={...(current[gate]||{}),[key]:box.checked};
        schema.settings={...(schema.settings||{}),[KEY]:{...current,[gate]:forGate}};
        if(typeof window.xo==='function')window.xo(false);
        render();
      };
    });
    const reason=root.querySelector('#slq-scope-reason');
    if(reason)reason.oninput=()=>{
      const prev=schema.settings?.[REASONS]||{};
      schema.settings={...(schema.settings||{}),[REASONS]:{...prev,[gate]:reason.value}};
      if(typeof window.xo==='function')window.xo(false);
    };
